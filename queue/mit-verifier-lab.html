export function validateIso6346Container(raw: string): { valid: boolean; expected: number; got: number } {
  const clean = raw.replace(/[^A-Z0-9]/gi, "").toUpperCase();
  if (clean.length !== 11) return { valid: false, expected: -1, got: -1 };
  let sum = 0;
  for (let i = 0; i < 10; i++) {
    sum += getCharValue(clean[i]) * Math.pow(2, i);
  }
  const expected = (sum % 11) % 10;
  const got = parseInt(clean[10], 10);
  return { valid: expected === got, expected, got };
}
